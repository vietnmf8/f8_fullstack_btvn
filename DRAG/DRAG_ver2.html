# PHẦN 1: CHUẨN BỊ

# 1. Đối tượng Kéo và Thả

# Drag Items

  • Danh sách các đối tượng sẽ được kéo (ví dụ: A, B, C...)

# Drop Target

  • Danh sách các khu vực sẽ nhận đối tượng kéo (ví dụ: 1, 2, 3...).

  • Trên mỗi đối tượng này, tạo 3 trạng thái (States) với tên chính xác:

    Normal, Drop Correct, Drop Incorrect.

# Nút Submit

# Các layer phản hồi & các button

  • Correct (Khi trả lời đúng)
  • Incorrect (Khi trả lời sai và hết lượt)
  • TryAgain (Khi trả lời sai nhưng còn lượt)
  • Warning (Khi Drop Target chưa được lấp đầy)

# 2. Thiết lập Tương tác trên Slide (QUAN TRỌNG)

# Convert to Freeform Drag & Drop & Option

# Tạo Motion Path để Reset

  • Cách làm:
    1. Chọn một đối tượng kéo (ví dụ: A).
    1. Vào tab Animations -> Add Motion Path -> Lines.
    1. Trong Path Options, đặt Duration là 0.10 giây và Length là 0px.
  • Lưu ý: Lặp lại bước này cho TẤT CẢ các đối tượng kéo. Mỗi đối tượng sẽ có một motion path dài 0px của riêng nó.

# Tạo biến

# Biến Giá trị & Trạng thái: (Tạo cho mỗi Drag Item)

  • A, B... (Type: Number, Default: 0)
  • A_state, B_state... (Type: Text, Default: trống)

# Biến Điều khiển Chung

  • Completed (Type: True/False, Default: False)
  • Attempts (Type: Number, Default: 0)
  • Feedback (Type: Text, Default: trống)

# Biến cho Scrolling Panel (Chỉ tạo khi cần)

  • DOMState_A, DOMState_B... (Type: Text, Default: trống)

window.SL_SaveInitialDOMStates = function() {
    var player = GetPlayer();
    var itemsToSave = window.SL_MainConfig.items;
    for (var itemKey in itemsToSave) {
        if (itemsToSave.hasOwnProperty(itemKey)) {
            var itemConfig = itemsToSave[itemKey];
            if (itemConfig.dataModelId && itemConfig.domStateVar) {
                const shape = document.querySelector(`[data-model-id="${itemConfig.dataModelId}"]`);
                const container = document.getElementById("acc-" + itemConfig.dataModelId);
                if (container && shape) {
                    const containerStyle = window.getComputedStyle(container);
                    const shapeStyle = window.getComputedStyle(shape);
                    const initialState = {
                        containerState: { position: containerStyle.position, left: containerStyle.left, top: containerStyle.top, transform: containerStyle.transform },
                        shapeState: { transform: shapeStyle.transform }
                    };
                    player.SetVar(itemConfig.domStateVar, JSON.stringify(initialState));
                }
            }
        }
    }
};

window.SL_ResetInteraction = function() {
    var player = GetPlayer();
    var config = window.SL_MainConfig;
    var items = config.items;
    var fbConfig = config.feedbackConfig;
    var itemStateNames = config.advancedConfig.itemStates;
    var domConfig = config.advancedConfig.dom;
    for (var itemKey in items) {
        if (items.hasOwnProperty(itemKey)) {
            var itemConfig = items[itemKey];
            player.SetVar(itemConfig.storylineVar, 0);
            if (itemConfig.stateVarName) {
                player.SetVar(itemConfig.stateVarName, "");
                player.SetVar(itemConfig.stateVarName, itemStateNames.normal);
            }
        }
    }
    player.SetVar(fbConfig.completionVar, false);

    // CẬP NHẬT MỚI: Tự động xóa biến Feedback để reset hoàn toàn
    if (typeof window.SL_ClearFeedbackState === 'function') {
        window.SL_ClearFeedbackState();
    }

    if (config.interactionModes.isScrollingPanel && domConfig && domConfig.scrollingPanelId) {
        const scrollPanel = document.querySelector(`[data-model-id="${domConfig.scrollingPanelId}"]`);
        const scrollAreaContent = scrollPanel ? scrollPanel.querySelector('.scrollarea-area.is-scrollable > div') : null;
        if (!scrollAreaContent) { return; }
        for (var itemKey in items) {
            if (items.hasOwnProperty(itemKey)) {
                var itemConfig = items[itemKey];
                if (itemConfig.dataModelId && itemConfig.domStateVar) {
                    var savedStateString = player.GetVar(itemConfig.domStateVar);
                    if (savedStateString) {
                        try {
                            const savedState = JSON.parse(savedStateString);
                            const shape = document.querySelector(`[data-model-id="${itemConfig.dataModelId}"]`);
                            const container = document.getElementById("acc-" + itemConfig.dataModelId);
                            if (container && shape) {
                                scrollAreaContent.appendChild(container);
                                scrollAreaContent.appendChild(shape);
                                const cState = savedState.containerState;
                                container.style.position = cState.position;
                                container.style.left = cState.left;
                                container.style.top = cState.top;
                                container.style.transform = cState.transform;
                                const sState = savedState.shapeState;
                                shape.style.transform = sState.transform;
                            }
                        } catch (e) {}
                    }
                }
            }
        }
    }
};

window.SL_ResetIncorrect_VariablesOnly = function() {
    var player = GetPlayer();
    var config = window.SL_MainConfig;
    var items = config.items;
    for (var itemKey in items) {
        if (items.hasOwnProperty(itemKey)) {
            var itemConfig = items[itemKey];
            var currentItemValue = player.GetVar(itemConfig.storylineVar);
            if (currentItemValue === 0) {
                continue;
            }
            var isCorrect = false;
            if (config.interactionModes.validationMode === 'individual') {
                var correctGroupKey = itemConfig.correctGroup;
                if (correctGroupKey && config.targetGroups[correctGroupKey] && config.targetGroups[correctGroupKey].includes(currentItemValue)) {
                    isCorrect = true;
                }
            }
            if (!isCorrect) {
                player.SetVar(itemConfig.storylineVar, 0);
            }
        }
    }
};

window.SL_HandleDrop = function(draggedItemKey, droppedOnTargetKey) {
    var player = GetPlayer();
    var config = window.SL_MainConfig;
    var items = config.items;
    var targets = config.targets;
    var interactionModes = config.interactionModes;
    var itemStateNames = config.advancedConfig.itemStates;
    var draggedItemConfig = items[draggedItemKey];
    if (!draggedItemConfig) return;
    var newValueForDraggedItem = 0;
    var itemVisualState = itemStateNames.normal;
    if (droppedOnTargetKey) {
        var targetConfig = targets[droppedOnTargetKey];
        if (targetConfig) {
            newValueForDraggedItem = targetConfig.value;
            if (interactionModes.dropFeedbackMode === 'instant') {
                var isCorrectDrop = false;
                if (interactionModes.validationMode === 'individual') {
                    var correctGroupKey = draggedItemConfig.correctGroup;
                    if (correctGroupKey && config.targetGroups[correctGroupKey] && config.targetGroups[correctGroupKey].includes(newValueForDraggedItem)) {
                        isCorrectDrop = true;
                    }
                } else {
                    var belongingSet = config.validationSets.find(function(set) { return set.targetValues.includes(newValueForDraggedItem); });
                    if (belongingSet && belongingSet.requiredItems.includes(draggedItemKey)) {
                        isCorrectDrop = true;
                    }
                }
                itemVisualState = isCorrectDrop ? itemStateNames.dropCorrect : itemStateNames.dropIncorrect;
            } else {
                itemVisualState = itemStateNames.normal;
            }
        }
    }
    player.SetVar(draggedItemConfig.storylineVar, newValueForDraggedItem);
    if (draggedItemConfig.stateVarName) {
        player.SetVar(draggedItemConfig.stateVarName, "");
        player.SetVar(draggedItemConfig.stateVarName, itemVisualState);
    }
    if (droppedOnTargetKey && newValueForDraggedItem !== 0) {
        for (var otherItemKey in items) {
            if (otherItemKey === draggedItemKey) continue;
            var otherItemConfig = items[otherItemKey];
            if (player.GetVar(otherItemConfig.storylineVar) === newValueForDraggedItem) {
                player.SetVar(otherItemConfig.storylineVar, 0);
                if (otherItemConfig.stateVarName) {
                    player.SetVar(otherItemConfig.stateVarName, "");
                    player.SetVar(otherItemConfig.stateVarName, itemStateNames.normal);
                }
            }
        }
    }
    if (typeof window.SL_CheckDragDropCompletion === 'function') {
        window.SL_CheckDragDropCompletion();
    }
};

window.SL_RelayDropEvent = function(targetKey) {
    setTimeout(function() {
        var config = window.SL_MainConfig;
        var items = config.items;
        var foundItemKey = null;
        var highestZIndex = -1;
        for (var itemKey in items) {
            if (items.hasOwnProperty(itemKey)) {
                var itemConfig = items[itemKey];
                var element = document.querySelector('[data-model-id="' + itemConfig.dataModelId + '"]');
                if (element && element.style.zIndex) {
                    var currentZIndex = parseInt(element.style.zIndex, 10);
                    if (currentZIndex > highestZIndex) {
                        highestZIndex = currentZIndex;
                        foundItemKey = itemKey;
                    }
                }
            }
        }
        if (foundItemKey) {
            window.SL_HandleDrop(foundItemKey, targetKey);
        } else {
            console.log("SL_RelayDropEvent: Không tìm thấy đối tượng nào có z-index cao nhất.");
        }
    }, 10);
};

window.SL_CheckDragDropCompletion = function() {
    var player = GetPlayer();
    var config = window.SL_MainConfig;
    if (!config.feedbackConfig || !config.feedbackConfig.completionVar) return;
    var allRequiredTargetsAreFilled = true;
    var targetsToCheck = new Set();
    if (config.interactionModes.validationMode === 'set') {
        config.validationSets.forEach(function(set) { set.targetValues.forEach(function(val) { targetsToCheck.add(val); }); });
    } else {
        for (var targetKey in config.targets) { targetsToCheck.add(config.targets[targetKey].value); }
    }
    if (targetsToCheck.size > 0) {
        targetsToCheck.forEach(function(targetValue) {
            var isThisTargetOccupied = false;
            for (var itemKey in config.items) {
                if (player.GetVar(config.items[itemKey].storylineVar) === targetValue) { isThisTargetOccupied = true; break; }
            }
            if (!isThisTargetOccupied) { allRequiredTargetsAreFilled = false; }
        });
    } else {
        allRequiredTargetsAreFilled = true;
    }
    player.SetVar(config.feedbackConfig.completionVar, allRequiredTargetsAreFilled);
};

window.SL_UpdateAllItemStates = function() {
    var player = GetPlayer();
    var config = window.SL_MainConfig;
    var itemStateNames = config.advancedConfig.itemStates;
    for (var itemKey in config.items) {
        if (config.items.hasOwnProperty(itemKey)) {
            var itemConfig = config.items[itemKey];
            var currentItemValue = player.GetVar(itemConfig.storylineVar);
            var itemVisualState = itemStateNames.normal;
            if (currentItemValue !== 0) {
                var isCorrectDrop = false;
                if (config.interactionModes.validationMode === 'individual') {
                    var correctGroupKey = itemConfig.correctGroup;
                    if (correctGroupKey && config.targetGroups[correctGroupKey] && config.targetGroups[correctGroupKey].includes(currentItemValue)) {
                        isCorrectDrop = true;
                    }
                } else {
                    var belongingSet = config.validationSets.find(function(set) { return set.requiredItems.includes(itemKey); });
                    if (belongingSet && belongingSet.targetValues.includes(currentItemValue)) {
                        isCorrectDrop = true;
                    }
                }
                itemVisualState = isCorrectDrop ? itemStateNames.dropCorrect : itemStateNames.dropIncorrect;
            }
            if (itemConfig.stateVarName) {
                player.SetVar(itemConfig.stateVarName, "");
                player.SetVar(itemConfig.stateVarName, itemVisualState);
            }
        }
    }
};

window.SL_ProcessFinalAnswer = function() {
    var player = GetPlayer();
    var config = window.SL_MainConfig;
    var fbConfig = config.feedbackConfig;
    if (!fbConfig) return;
    var isAnswerFullyCorrect = true;
    if (config.interactionModes.validationMode === 'set') {
        var allItemsInSets = new Set();
        for (var i = 0; i < config.validationSets.length; i++) {
            var set = config.validationSets[i];
            var requiredItemsSet = new Set(set.requiredItems);
            var foundItems = [];
            set.requiredItems.forEach(function(itemKey) { allItemsInSets.add(itemKey); });
            for (var itemKey in config.items) {
                var itemValue = player.GetVar(config.items[itemKey].storylineVar);
                if (set.targetValues.includes(itemValue)) { foundItems.push(itemKey); }
            }
            var foundItemsSet = new Set(foundItems);
            if (requiredItemsSet.size !== foundItemsSet.size) { isAnswerFullyCorrect = false; break; }
            requiredItemsSet.forEach(function(item) { if (!foundItemsSet.has(item)) { isAnswerFullyCorrect = false; } });
            if (!isAnswerFullyCorrect) break;
        }
        if (isAnswerFullyCorrect) {
            for (var itemKey in config.items) {
                if (!allItemsInSets.has(itemKey) && player.GetVar(config.items[itemKey].storylineVar) !== 0) { isAnswerFullyCorrect = false; break; }
            }
        }
    } else {
        for (var itemKey in config.items) {
            var itemConfig = config.items[itemKey];
            var currentItemValue = player.GetVar(itemConfig.storylineVar);
            var correctGroupKey = itemConfig.correctGroup;
            if (correctGroupKey) {
                if (!config.targetGroups[correctGroupKey] || !config.targetGroups[correctGroupKey].includes(currentItemValue)) { isAnswerFullyCorrect = false; break; }
            } else {
                if (currentItemValue !== 0) { isAnswerFullyCorrect = false; break; }
            }
        }
    }
    if (typeof window.SL_UpdateAllItemStates === 'function') {
        window.SL_UpdateAllItemStates();
    }
    if (isAnswerFullyCorrect) {
        player.SetVar(fbConfig.feedbackVar, fbConfig.correctLayer);
    } else {
        var currentAttempts = player.GetVar(fbConfig.attemptsVar);
        var maxAttempts = fbConfig.maxAttempts || 1;
        currentAttempts++;
        player.SetVar(fbConfig.attemptsVar, currentAttempts);
        if (currentAttempts >= maxAttempts) {
            player.SetVar(fbConfig.feedbackVar, fbConfig.incorrectLayer);
        } else {
            player.SetVar(fbConfig.feedbackVar, fbConfig.tryAgainLayer);
        }
    }
};

// HÀM TRUNG GIAN XỬ LÝ SỰ KIỆN CLICK NÚT "NEXT"

window.SL_HandleNextButtonClick = function() {
    var player = GetPlayer();
    var config = window.SL_MainConfig;
    if (!config.feedbackConfig) return;
    if (config.interactionModes.submitButtonInitialMode === "visible") {
        var isDragDropConsideredComplete = player.GetVar(config.feedbackConfig.completionVar);
        if (!isDragDropConsideredComplete) {
            if (config.feedbackConfig.warningLayer && config.feedbackConfig.feedbackVar) {
                player.SetVar(config.feedbackConfig.feedbackVar, config.feedbackConfig.warningLayer);
            }
            return;
        }
    }
    if (typeof window.SL_ProcessFinalAnswer === 'function') {
        window.SL_ProcessFinalAnswer();
    }
};

    }
};

  • Action: Execute JavaScript
window.SL_RelayDropEvent("2");
  • When: Object Dropped On
  • Object: A
  • Target: 2

  • Object: A
